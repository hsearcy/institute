# M3: 3D Hub + Placeholder Tutor — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the Institute kid-app's first on-device 3D scene — a low-poly hub world with a placeholder tutor and 4 placeholder mini-game portals — running at ≥30fps on Fire HD 10.

**Architecture:** A new Expo app (`apps/kid-app/`) joins the monorepo. R3F renders a single 3D `<Canvas>` (HubScene). Pure logic (camera math, FPS measurement, asset state) lives in unit-testable modules; 3D components are thin wrappers around primitives. RN UI overlays handle loading + FPS + toast. Smoke-tested in headless mode; visually verified on a real Fire HD 10.

**Tech Stack:** Expo SDK (latest stable, ~52+), React Native, react-three-fiber (native build), @react-three/drei, expo-gl, expo-asset, TypeScript, Bun (workspaces + test runner).

## Pre-flight notes for the implementer

- **Bun + Expo monorepo:** Expo's Metro bundler needs help finding hoisted workspace deps. The plan includes a `metro.config.js` that sets `watchFolders` to the repo root and disables hierarchical module lookup — copy it verbatim. Do NOT add `nohoist` to the root `package.json` — Bun's default hoisting works fine with the Metro config.
- **R3F native vs web:** Import from `@react-three/fiber/native` for the `Canvas`, NOT from `@react-three/fiber`. The web version will silently render nothing in Expo. Same applies to drei: import from `@react-three/drei/native` where available; some helpers only exist in `@react-three/drei` and that's fine — TS will tell you.
- **Expo Web for fast iteration:** This plan's primary dev loop is `bun expo start --web`, with periodic on-device passes via Expo Go on the Fire tablet. Browser DOM works for ~95% of code; the perf budget verification requires the actual tablet.
- **GL context: a single `<Canvas>` only.** Do not nest. Multiple GL contexts are not supported on Fire OS.

```ts
// apps/kid-app/src/world/camera.ts

/** Spherical → cartesian camera placement, with hero at origin. */
export type CameraSpec = {
  /** Distance from hero, in world units. */
  distance: number;
  /** Elevation angle from horizontal, in radians (0 = horizon, π/2 = directly above). */
  pitch: number;
  /** Horizontal angle around hero, in radians (0 = south of hero, +π/2 = east). */
  yaw: number;
};

/** Maximum yaw deviation from the default (south-facing) orientation. */
export const YAW_RANGE_RAD = Math.PI / 6; // 30°

/** Clamp yaw to [-YAW_RANGE_RAD, YAW_RANGE_RAD]. */
export function clampYaw(yaw: number): number {
  if (yaw > YAW_RANGE_RAD) return YAW_RANGE_RAD;
  if (yaw < -YAW_RANGE_RAD) return -YAW_RANGE_RAD;
  return yaw;
}

/** Compute the camera world position given a CameraSpec. */
export function cameraPosition(spec: CameraSpec): readonly [number, number, number] {
  const horizontal = spec.distance * Math.cos(spec.pitch);
  const x = horizontal * Math.sin(spec.yaw);
  const y = spec.distance * Math.sin(spec.pitch);
  const z = horizontal * Math.cos(spec.yaw);
  return [x, y, z] as const;
}
```

```ts
// apps/kid-app/src/world/hooks/useFps.ts
import { useRef, useState } from "react";

export const FPS_WINDOW = 60;

/**
 * Pure: given a list of frame-end timestamps (ms), compute current FPS.
 * Returns 0 if too few samples to measure.
 */
export function computeFps(stamps: ReadonlyArray<number>): number {
  if (stamps.length < 2) return 0;
  const first = stamps[0]!;
  const last = stamps[stamps.length - 1]!;
  const elapsedMs = last - first;
  if (elapsedMs <= 0) return 0;
  const frames = stamps.length - 1;
  return Math.round((frames * 1000) / elapsedMs);
}

/**
 * React hook for frame-time tracking. Pushes timestamps from R3F's
 * useFrame into a rolling buffer; updates the visible FPS at most every
 * 10 frames so the overlay doesn't churn the React tree on every frame.
 *
 * Wire as:
 *   const fps = useFps();
 *   useFrame(({ clock }) => fps.tick(clock.elapsedTime * 1000));
 *   // <Text>{fps.value} FPS</Text>
 */
export function useFps(): { value: number; tick: (nowMs: number) => void } {
  const buf = useRef<number[]>([]);
  const tickCount = useRef(0);
  const [value, setValue] = useState(0);

  const tick = (nowMs: number) => {
    buf.current.push(nowMs);
    if (buf.current.length > FPS_WINDOW) buf.current.shift();
    tickCount.current += 1;
    if (tickCount.current % 10 === 0) {
      setValue(computeFps(buf.current));
    }
  };

  return { value, tick };
}
```

```ts
// apps/kid-app/src/world/hooks/useAssets.ts
import { useReducer } from "react";

export type AssetState =
  | { status: "idle" }
  | { status: "loading" }
  | { status: "ready" }
  | { status: "error"; message: string };

export type AssetEvent =
  | { type: "START" }
  | { type: "COMPLETE" }
  | { type: "FAIL"; message: string }
  | { type: "RESET" };

export const initial: AssetState = { status: "idle" };

export function reduce(state: AssetState, event: AssetEvent): AssetState {
  switch (event.type) {
    case "RESET":
      return initial;
    case "START":
      if (state.status === "idle") return { status: "loading" };
      return state;
    case "COMPLETE":
      if (state.status === "loading") return { status: "ready" };
      return state;
    case "FAIL":
      if (state.status === "loading") return { status: "error", message: event.message };
      return state;
    default:
      return state;
  }
}

/** React hook wrapper around the reducer. Used by the LoadingScreen + App. */
export function useAssets() {
  const [state, dispatch] = useReducer(reduce, initial);
  return {
    state,
    start: () => dispatch({ type: "START" }),
    complete: () => dispatch({ type: "COMPLETE" }),
    fail: (message: string) => dispatch({ type: "FAIL", message }),
    reset: () => dispatch({ type: "RESET" }),
  };
}
```

```tsx
import { useFrame } from "@react-three/fiber/native";
import { Text } from "@react-three/drei/native";
import type { Mesh } from "three";
import type { PortalDef } from "../../config";

type Props = {
  def: PortalDef;
  /** Phase offset in seconds so portals don't pulse in unison. */
  phaseOffsetS?: number;
  onTap?: (id: PortalDef["id"]) => void;
};

const PULSE_PERIOD_S = 1.0;
const PULSE_AMP = 0.02;

export function Portal({ def, phaseOffsetS = 0, onTap }: Props) {
  const ref = useRef<Mesh>(null);
  useFrame(({ clock }) => {
    if (!ref.current) return;
    const t = clock.elapsedTime + phaseOffsetS;
    const s = 1 + Math.sin((2 * Math.PI * t) / PULSE_PERIOD_S) * PULSE_AMP;
    ref.current.scale.set(s, s, s);
  });
  return (
    <group position={def.position}>
      <mesh
        ref={ref}
        position={[0, 1.2, 0]}
        onClick={(e) => {
          e.stopPropagation();
          onTap?.(def.id);
```

```tsx
export function Toast({ message, onHide }: { message: string; onHide: () => void }) {
  const opacity = useAnimatedValue(0);

  useEffect(() => {
    Animated.sequence([
      Animated.timing(opacity, { toValue: 1, duration: 200, useNativeDriver: true }),
      Animated.delay(1600),

    ]).start(() => onHide());
  }, [message]);
```

```tsx
// apps/kid-app/src/world/SceneController.tsx
import { useFrame, useThree } from "@react-three/fiber/native";
import { useEffect } from "react";
import { cameraPosition, clampYaw } from "./camera";
import { useFps } from "./hooks/useFps";

type Props = {
  yaw: number;
  pitch: number;
  distance: number;
  onFps: (fps: number) => void;
};

export function SceneController({ yaw, pitch, distance, onFps }: Props) {
  const { camera } = useThree();
  const fps = useFps();

  useEffect(() => {
    const [x, y, z] = cameraPosition({ yaw: clampYaw(yaw), pitch, distance });
    camera.position.set(x, y, z);
    camera.lookAt(0, 1.2, 0); // look at hero's chest
  }, [yaw, pitch, distance, camera]);

  useFrame(({ clock }) => {
    fps.tick(clock.elapsedTime * 1000);
  });

  useEffect(() => {
    onFps(fps.value);
  }, [fps.value, onFps]);

  return null;
}
```

```tsx
    const cylinders = renderer.scene.findAll(

        node.type === "Mesh" &&
        // @ts-expect-error - geometry presence checked at runtime
        node.instance?.geometry?.type === "CylinderGeometry"
    );

    expect(cylinders.length).toBeGreaterThanOrEqual(5);
    await renderer.unmount();

  test("portal tap fires onPortalTap with the right id", async () => {

    const renderer = await ReactThreeTestRenderer.create(
      <HubScene onPortalTap={(id) => { received = id; }} />,
    );
    // Find the first portal cylinder mesh
    const portalMeshes = renderer.scene.findAll(
      (node) =>
        node.type === "Mesh" &&
        // @ts-expect-error - geometry presence checked at runtime

        node.instance?.material?.emissive !== undefined

    await renderer.unmount();
```
