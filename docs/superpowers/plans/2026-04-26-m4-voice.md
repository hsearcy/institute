# M4 — Voice In/Out Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Sam plays Spell Cast hands-free — tutor's intro spoken via ElevenLabs, his answers transcribed via Deepgram (streaming WebSocket), all flowing through the existing Rails backend. NumberPad remains as the always-available fallback.

**Architecture:** Backend gains: ActionCable (re-added, was skipped in M5), `Tts::Stream` (ElevenLabs streaming HTTP), `TtsController` (chunked-response streaming via `ActionController::Live`), `SttChannel` (ActionCable, wraps `Stt::DeepgramSession` which proxies to Deepgram's streaming WS), and `Tutor::SpeechNormalizer` (transcript → integer with toddler-isms + expected-answer fuzzy matching). Kid-app gains: `expo-audio` for capture+playback, `@rails/actioncable` JS client, a small voice subsystem under `src/voice/` (recorder, sttSession, ttsClient, voiceFlow, permission), and two UI primitives (`MicIcon`, `Confirmation`). `MiniGameMount` orchestrates the speak-then-listen flow when mic permission is granted; falls back silently to `NumberPad` on any failure.

**Tech Stack:** Rails 8.1 (Ruby 3.4.7), ActionCable + solid_cable, `faye-websocket` (Deepgram WS client), `Faraday` (ElevenLabs HTTP), Minitest. TypeScript, pnpm, Vitest, `expo-audio` (Expo SDK 54+), `@rails/actioncable`, react-native.

    def connect
      secret = ENV["TUTOR_SHARED_SECRET"]

      reject_unauthorized_connection unless ActiveSupport::SecurityUtils.secure_compare(header, "Bearer #{secret}")

module Tutor
  module SpeechNormalizer
    NUMBER_WORDS = {
      "zero" => 0, "one" => 1, "two" => 2, "three" => 3, "four" => 4,
      "five" => 5, "six" => 6, "seven" => 7, "eight" => 8, "nine" => 9,
      "ten" => 10, "eleven" => 11, "twelve" => 12, "thirteen" => 13,
      "fourteen" => 14, "fifteen" => 15, "sixteen" => 16, "seventeen" => 17,
      "eighteen" => 18, "nineteen" => 19, "twenty" => 20,
      "thirty" => 30, "forty" => 40, "fifty" => 50, "sixty" => 60,
      "seventy" => 70, "eighty" => 80, "ninety" => 90, "hundred" => 100,
    }.freeze

    # Common kid-speech mappings to canonical number words. STT will frequently
    # return these for actual digits the kid said.
    TODDLER_ISMS = {
      "free" => "three", "tree" => "three", "twee" => "three", "fwee" => "three",
      "fife" => "five", "fwive" => "five",
      "tin" => "ten",
      "noyne" => "nine",
    }.freeze

    def self.call(transcript, expected_answers)
      n = parse(transcript)
      return { transcript: transcript, normalized: nil, confidence: "no_match" } if n.nil?
      conf = expected_answers.include?(n) ? "high" : "low"
      { transcript: transcript, normalized: n, confidence: conf }
    end

    def self.parse(raw)
      return nil if raw.nil?
      cleaned = raw.downcase.gsub(/[^a-z0-9\s]/, " ").squeeze(" ").strip
      return nil if cleaned.empty?

      tokens = cleaned.split(" ").map { |t| TODDLER_ISMS.fetch(t, t) }

      # Direct integer literal anywhere in the tokens
      tokens.each do |t|
        return Integer(t) if t.match?(/\A\d+\z/)
      end

      # Word-numbers: handle "twenty one" / "one hundred" by accumulating
      sum = 0; current = 0; found = false
      tokens.each do |t|
        if (n = NUMBER_WORDS[t])
          found = true
          if n == 100
            current = (current.zero? ? 1 : current) * 100
          elsif n >= 20
            current += n
          else
            current += n
          end
        elsif found && current > 0
          # word-number sequence ended; capture and reset for the next one
          sum += current; current = 0; found = false
        end
      end
      sum += current
      found || sum > 0 ? sum : nil
    end
  end
end
```

require "faraday"

module Tts
  module Stream
    DEFAULT_VOICE_ID = ENV.fetch("ELEVENLABS_VOICE_ID", "21m00Tcm4TlvDq8ikWAM")
    DEFAULT_MODEL_ID = "eleven_turbo_v2_5"
    BASE = "https://api.elevenlabs.io".freeze

    class StreamError < StandardError
      attr_reader :status, :body
      def initialize(status, body)
        @status = status
        @body = body
        super("ElevenLabs returned #{status}: #{body}")
      end
    end

    # Default real HTTP client wrapping Faraday with streamed body callback.
    module RealHttp
      def self.post(url, body, headers, &on_chunk)
        conn = Faraday.new do |f|
          f.adapter :net_http   # default adapter; supports on_data callback below
        end
        response = conn.post(url) do |req|
          req.headers.merge!(headers)
          req.body = body
          # Faraday net_http adapter supports `on_data` for chunked responses
          req.options.on_data = proc { |chunk, _| on_chunk.call(chunk) if chunk && !chunk.empty? }
        end
        raise StreamError.new(response.status, response.body) if response.status >= 400
      end
    end

    def self.call(text:, voice_id: DEFAULT_VOICE_ID, http: RealHttp, &on_chunk)
      url = "#{BASE}/v1/text-to-speech/#{voice_id}/stream"
      headers = {
        "xi-api-key"   => ENV["ELEVENLABS_API_KEY"].to_s,
        "accept"       => "audio/mpeg",
        "content-type" => "application/json",
      }
      body = JSON.generate(
        text: text,
        model_id: DEFAULT_MODEL_ID,
        voice_settings: { stability: 0.5, similarity_boost: 0.7 },
      )
      http.post(url, body, headers, &on_chunk)
    end
  end
end
```

  include ActionController::Live
  skip_before_action :verify_authenticity_token, raise: false
  before_action :authenticate_bearer!

  class << self
    attr_accessor :tts_stream
  end
  self.tts_stream = Tts::Stream

  def create
    text = params[:text].to_s
    if text.empty?
      return render json: { error: "missing text" }, status: :unprocessable_entity
    end
    voice_id = params[:voiceId].presence

    response.headers["Content-Type"] = "audio/mpeg"
    response.headers["Cache-Control"] = "no-cache"

    begin
      if voice_id
        self.class.tts_stream.call(text: text, voice_id: voice_id) { |c| response.stream.write(c) }
      else
        self.class.tts_stream.call(text: text) { |c| response.stream.write(c) }
      end
    rescue Tts::Stream::StreamError => e
      Rails.logger.warn("Tts::Stream error: #{e.message}")
      response.stream.close unless response.stream.closed?
      # Once headers are sent we can't change status — but this rescue only
      # fires before the first chunk if ElevenLabs returns 4xx/5xx, so we
      # can still set status before the first write.
      self.response.status = 502
    rescue => e
      Rails.logger.error("TtsController error: #{e.class} #{e.message}")
      self.response.status = 500 unless response.committed?
    ensure
      response.stream.close unless response.stream.closed?
    end
  end

  private

  def authenticate_bearer!
    secret = ENV["TUTOR_SHARED_SECRET"]
    return if secret.nil? || secret.empty?
    header = request.headers["Authorization"].to_s

module Stt
  class DeepgramSession
    DEEPGRAM_WS = "wss://api.deepgram.com/v1/listen?" \
                  "model=nova-2&endpointing=300&interim_results=false&" \
                  "language=en&punctuate=false&numerals=true&encoding=linear16&sample_rate=16000".freeze

    def initialize(expected_answers:, ws: nil, &on_event)
      @expected_answers = expected_answers
      @on_event = on_event
      @ws = ws  # injectable for tests; nil means we'll build a real one in start
      @open = false
      @buffer = []
    end

    def start
      @ws ||= build_real_ws
      @ws.on(:open) do
        @open = true
        @buffer.each { |b| @ws.send(b) }
        @buffer.clear
      end
      @ws.on(:message) { |event| handle_message(event.data) }
      @ws.on(:close) do
        @open = false
      end
      self
    end

    def send_audio(frame)
      if @open
        @ws.send(frame)
      else
        @buffer << frame
      end
    end

    def close
      @ws&.close
    end

    private

    def handle_message(data)
      payload = JSON.parse(data) rescue nil
      return unless payload
      return unless payload["is_final"] && payload["speech_final"]
      transcript = payload.dig("channel", "alternatives", 0, "transcript").to_s
      return if transcript.strip.empty?
      result = Tutor::SpeechNormalizer.call(transcript, @expected_answers)
      @on_event&.call(result)
    end

    def build_real_ws
      require "faye/websocket"
      headers = { "Authorization" => "Token #{ENV.fetch('DEEPGRAM_API_KEY')}" }
      Faye::WebSocket::Client.new(DEEPGRAM_WS, [], headers: headers)
    end
  end
end

// expo-audio's playback API in SDK 54 exposes either createAudioPlayer or
// useAudioPlayer (hook). For an imperative speak(), we use the non-hook form.
// Verify the actual export when wiring on-device — the mock in the test
// matches the *intended* shape (createAudioPlayer + addListener).
import { createAudioPlayer } from "expo-audio";

export type SpeakOpts = {
  baseUrl: string;
  bearerToken?: string;
  text: string;
  voiceId?: string;
};

/**
 * Fetches TTS audio for `text` and plays it through the device speaker.
 * Resolves when playback completes. Rejects on HTTP error.
 *
 * No-op (resolves immediately) when baseUrl is empty (offline / no backend).
 */
export async function speak({ baseUrl, bearerToken, text, voiceId }: SpeakOpts): Promise<void> {
  if (!baseUrl) return;

  const headers: Record<string, string> = { "Content-Type": "application/json" };
  if (bearerToken) headers["Authorization"] = `Bearer ${bearerToken}`;

  const res = await fetch(`${baseUrl}/tts`, {
    method: "POST",
    headers,
    body: JSON.stringify(voiceId ? { text, voiceId } : { text }),
  });
  if (!res.ok) throw new Error(`tts failed: ${res.status}`);

  const audioBytes = await res.arrayBuffer();

  // expo-audio plays from a URI; for an in-memory buffer we use a data URI.
  const b64 = arrayBufferToBase64(audioBytes);
  const player = createAudioPlayer({ uri: `data:audio/mpeg;base64,${b64}` });

  await new Promise<void>((resolve) => {
    player.addListener("playbackStatusUpdate", (status: { didJustFinish?: boolean } = {}) => {
      if (status.didJustFinish) resolve();
    });
    player.play();
  });
}

function arrayBufferToBase64(buf: ArrayBuffer): string {
  // RN-safe base64 encode via global btoa + binary string
  let bin = "";
  const bytes = new Uint8Array(buf);
  for (let i = 0; i < bytes.length; i++) bin += String.fromCharCode(bytes[i]!);
  // global btoa exists in RN (and Node 18+)
  return globalThis.btoa(bin);
}

import { createConsumer } from "@rails/actioncable";

export type SttResult = {
  transcript: string;
  normalized: number | null;
  confidence: "high" | "low" | "no_match";
};

export type StartSttSessionOpts = {
  wsUrl: string;
  bearerToken?: string;
  expectedAnswers: number[];
  onResult: (r: SttResult) => void;
  onError: (e: Error) => void;
};

export type SttHandle = {
  sendAudio: (chunk: ArrayBuffer) => void;
  stop: () => void;
};

const INERT: SttHandle = { sendAudio: () => {}, stop: () => {} };

export function startSttSession(opts: StartSttSessionOpts): SttHandle {
  if (!opts.wsUrl) return INERT;

  const url = opts.bearerToken ? `${opts.wsUrl}?token=${encodeURIComponent(opts.bearerToken)}` : opts.wsUrl;
  const consumer = createConsumer(url);

  const sub = consumer.subscriptions.create(
    { channel: "SttChannel", expected_answers: opts.expectedAnswers },
    {
      received: (data: unknown) => {
        if (data && typeof data === "object") opts.onResult(data as SttResult);
      },
      disconnected: () => opts.onError(new Error("STT WS disconnected")),
    },
  );

  return {
    sendAudio: (chunk) => { sub.send(chunk); },
    stop: () => { sub.unsubscribe(); consumer.disconnect(); },
  };
}

import type { SttResult } from "./sttSession";

export type VoiceFlowDeps = {
  speak: (text: string) => Promise<void>;
  startRecorder: (onChunk: (buf: ArrayBuffer) => void) => Promise<{ stop: () => Promise<void> }>;
  startSttSession: (opts: {
    expectedAnswers: number[];
    onResult: (r: SttResult) => void;
    onError: (e: Error) => void;
  }) => { sendAudio: (chunk: ArrayBuffer) => void; stop: () => void };
  submitAnswer: (n: number) => void;
  onLowConfidence: (candidate: number, expected: number[]) => void;
  onNoMatch: () => void;
};

export type RunVoiceFlowOpts = {
  intro: string;
  expectedAnswers: number[];
  deps: VoiceFlowDeps;
};

export type VoiceFlowHandle = {
  stop: () => Promise<void>;
};

/**
 * Per-step voice flow:
 *   1. Speak the intro (mic muted during playback)
 *   2. Open recorder + STT session
 *   3. Forward chunks; route results by confidence
 */
export async function runVoiceFlow(opts: RunVoiceFlowOpts): Promise<VoiceFlowHandle> {
  await opts.deps.speak(opts.intro);

  const session = opts.deps.startSttSession({
    expectedAnswers: opts.expectedAnswers,
    onResult: (r) => {
      switch (r.confidence) {
        case "high":
          if (r.normalized !== null) opts.deps.submitAnswer(r.normalized);
          break;
        case "low":
          if (r.normalized !== null) opts.deps.onLowConfidence(r.normalized, opts.expectedAnswers);
          break;
        case "no_match":
          opts.deps.onNoMatch();
          break;
      }
    },
    onError: () => opts.deps.onNoMatch(),
  });

  const recorder = await opts.deps.startRecorder((chunk) => session.sendAudio(chunk));

  return {
    stop: async () => {
      await recorder.stop();
      session.stop();
    },
  };
}

Identify the right pattern. Common shapes in expo-audio for SDK 54:
- `useAudioRecorder(preset)` hook returning a recorder with `record()` / `stop()`
- `AudioRecorder` class with `prepareToRecordAsync` / `startAsync` / `stopAsync` / on-buffer event subscription
- Some versions don't expose chunk callbacks at all — in which case fall back to expo-av or use file-poll.

Pick the simplest path that supports streaming chunks to Deepgram. If chunk callbacks aren't available, the fallback is:
1. Record to an internal buffer file (default expo-audio behavior).
2. Periodically (every ~250ms) read the new bytes appended to that file.
3. Stream those bytes to the STT session.

This is laggier than true streaming but works.
